    });

    // 2) Bloqueia atalhos de teclado perigosos
    document.addEventListener('keydown', function(e) {
        // F12 — DevTools
        if (e.key === 'F12' || e.keyCode === 123) {
            e.preventDefault();
            return false;
        }
        // Ctrl+Shift+I / Ctrl+Shift+J / Ctrl+Shift+C — DevTools
        if (e.ctrlKey && e.shiftKey && (e.key === 'I' || e.key === 'J' || e.key === 'C' || e.keyCode === 73 || e.keyCode === 74 || e.keyCode === 67)) {
            e.preventDefault();
            return false;
        }
        // Ctrl+U — Ver código-fonte
        if (e.ctrlKey && (e.key === 'u' || e.key === 'U' || e.keyCode === 85)) {
            e.preventDefault();
            return false;
        }
        // Ctrl+S — Salvar página
        if (e.ctrlKey && (e.key === 's' || e.key === 'S' || e.keyCode === 83)) {
            e.preventDefault();
            return false;
        }
        // Ctrl+P — Imprimir
        if (e.ctrlKey && (e.key === 'p' || e.key === 'P' || e.keyCode === 80)) {
            e.preventDefault();
            return false;
        }
        // Ctrl+A — Selecionar tudo
        if (e.ctrlKey && (e.key === 'a' || e.key === 'A' || e.keyCode === 65)) {
            e.preventDefault();
            return false;
        }
        // Ctrl+C — Copiar
        if (e.ctrlKey && (e.key === 'c' || e.key === 'C' || e.keyCode === 67)) {
            // Permite copiar dentro do player (iframe não é afetado)
            const target = e.target;
            if (target && target.tagName && target.tagName.toLowerCase() === 'iframe') return;
            e.preventDefault();
            return false;
        }
        // Ctrl+Shift+S / Ctrl+Shift+E — Captura de tela no Firefox
        if (e.ctrlKey && e.shiftKey && (e.key === 'S' || e.key === 'E')) {
            e.preventDefault();
            return false;
        }
        // PrintScreen
        if (e.key === 'PrintScreen' || e.keyCode === 44) {
            // Não dá para impedir totalmente, mas tenta limpar a área de transferência
            try {
                navigator.clipboard.writeText('');
            } catch (err) {}
        }
    });

    document.addEventListener('dragstart', function(e) {
        e.preventDefault();
        return false;

function escapeHtml(str) {
    if (!str) return '';
    return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
}

async function tmdbFetch(endpoint, params = {}) {
    const url = new URL(`${TMDB_BASE}${endpoint}`);
    url.searchParams.set('api_key', CONFIG.API_KEY);
    url.searchParams.set('language', CONFIG.LANGUAGE);

    Object.entries(params).forEach(([key, value]) => {
        if (value !== undefined && value !== null) {
            url.searchParams.set(key, value);
        }
    });

    const response = await fetch(url.toString());

    if (!response.ok) {
        if (response.status === 401) {
            throw new Error('API Key inválida. Verifique sua chave do TMDB.');
        }
        throw new Error(`Erro na API: ${response.status}`);
    }

    return response.json();
}

async function fetchTVShow(tvId) {

        append_to_response: 'aggregate_credits,images'
    });
}

async function fetchSeason(tvId, seasonNumber) {
    return tmdbFetch(`/tv/${tvId}/season/${seasonNumber}`);
}

                    </div>
                </div>
            `;
        }).join('')
        : `<p class="empty-text">Elenco não disponível.</p>`;

    const seasons = (data.seasons || [])
        .filter(s => s.season_number > 0)
        .sort((a, b) => a.season_number - b.season_number);

    state.seasons = seasons;
    renderSeasonTabs(seasons, seasons[0]?.season_number || 1);

    dom.playerTitle.textContent = data.name || '';

    refreshIcons();
}

function renderSeasonTabs(seasons, activeSeason) {
    state.currentSeason = activeSeason;

    dom.seasonTabs.innerHTML = seasons.map(s => `
        <button class="season-tab ${s.season_number === activeSeason ? 'active' : ''}"
                data-season="${s.season_number}">
            <i data-lucide="folder" class="icon"></i>
            <span class="season-tab-text">
                Temporada ${s.season_number}
                <span class="episode-count">${s.episode_count} eps</span>
            </span>
        </button>
    `).join('');

    dom.seasonTabs.querySelectorAll('.season-tab').forEach(tab => {
        tab.addEventListener('click', () => {
            const seasonNum = parseInt(tab.dataset.season, 10);
            loadSeason(seasonNum);
        });
    });

    refreshIcons();
}

// TEMPORADAS
// ============================================================
async function loadSeason(seasonNumber) {
    state.currentSeason = seasonNumber;

    dom.seasonTabs.querySelectorAll('.season-tab').forEach(tab => {
        tab.classList.toggle('active', parseInt(tab.dataset.season, 10) === seasonNumber);
    });

    dom.currentSeasonTitle.textContent = `Temporada ${seasonNumber}`;
    dom.episodesList.innerHTML = '<div class="empty-state"><div class="spinner" style="width:40px;height:40px;border-width:3px;"></div><p>Carregando episódios...</p></div>';

    try {
        const seasonData = await fetchSeason(state.tvId, seasonNumber);
        state.episodes = seasonData.episodes || [];
        renderEpisodes(state.episodes, seasonNumber);
    } catch (error) {
        console.error('Erro ao carregar temporada:', error);
        dom.episodesList.innerHTML = `
            <div class="empty-state">
                <i data-lucide="alert-circle" class="icon"></i>
                <p>Erro ao carregar episódios. Tente novamente.</p>
            </div>
        `;
        refreshIcons();
    }
}

function renderEpisodes(episodes, seasonNumber) {
    if (!episodes.length) {
        dom.episodesList.innerHTML = `
            <div class="empty-state">
                <i data-lucide="inbox" class="icon"></i>
                <p>Nenhum episódio encontrado para esta temporada.</p>
            </div>
        `;
        refreshIcons();
        return;
    }

    dom.episodesList.innerHTML = episodes.map((ep, i) => {
        const thumb = ep.still_path
            ? `<img src="${getImageUrl(ep.still_path, 'w780')}" alt="${escapeHtml(ep.name)}" loading="lazy">`
            : `<div class="episode-thumb-placeholder"><i data-lucide="image-off" class="icon"></i></div>`;

        const dateStr = ep.air_date
            ? new Date(ep.air_date + 'T00:00:00').toLocaleDateString('pt-BR')
            : '';

        return `
            <div class="episode-card" style="animation-delay:${i * 0.05}s" data-episode="${ep.episode_number}" data-season="${seasonNumber}">
                <div class="episode-number">${ep.episode_number}</div>
                <div class="episode-thumb">${thumb}</div>
                <div class="episode-info">
                    <h4 class="episode-title">${escapeHtml(ep.name || `Episódio ${ep.episode_number}`)}</h4>
                    <p class="episode-overview">${escapeHtml(ep.overview || 'Sem descrição disponível.')}</p>
                    <div class="episode-meta">
                        ${dateStr ? `<span><i data-lucide="calendar" class="icon"></i>${dateStr}</span>` : ''}
                        ${ep.vote_average ? `<span class="episode-rating"><i data-lucide="star" class="icon"></i>${formatRating(ep.vote_average)}</span>` : ''}
                        ${ep.runtime ? `<span><i data-lucide="clock" class="icon"></i>${ep.runtime} min</span>` : ''}
                    </div>
                </div>
                <div class="episode-actions">
                    <button class="btn-watch" data-episode="${ep.episode_number}" data-season="${seasonNumber}">
                        <i data-lucide="play" class="icon"></i>
                        Assistir
                    </button>
                </div>
            </div>
        `;
    }).join('');

    dom.episodesList.querySelectorAll('.btn-watch').forEach(btn => {
        btn.addEventListener('click', (e) => {
            e.stopPropagation();
            const epNum = parseInt(btn.dataset.episode, 10);
            const seasonNum = parseInt(btn.dataset.season, 10);
            handleWatch(seasonNum, epNum);
        });
    });

    refreshIcons();
}

// ============================================================
// 🎥 HANDLE WATCH — REPRODUZ O EPISÓDIO (com HLS.js)
// ============================================================
function handleWatch(seasonNumber, episodeNumber) {
    const episode = state.episodes.find(ep => ep.episode_number === episodeNumber);
    if (!episode) return;

    const videoUrl = CONFIG.EPISODE_LINKS?.[seasonNumber]?.[episodeNumber];

    dom.episodesList.querySelectorAll('.episode-card').forEach(card => {
        const isActive =
            parseInt(card.dataset.episode, 10) === episodeNumber &&
            parseInt(card.dataset.season, 10) === seasonNumber;
        card.classList.toggle('active', isActive);
    });

    dom.playerTitle.textContent = `${state.tvData?.name || ''} — T${seasonNumber}E${episodeNumber}: ${episode.name}`;

    if (!videoUrl || videoUrl.trim() === '') {
        dom.playerContainer.classList.remove('playing');
        dom.playerPlaceholder.style.display = 'flex';
        dom.playerPlaceholder.innerHTML = `
            <div class="player-placeholder-inner">
                <div class="player-icon warning">
                    <i data-lucide="alert-triangle" class="icon"></i>
                </div>
                <h3>Link não configurado</h3>
                <p>Temporada ${seasonNumber} · Episódio ${episodeNumber}: ${escapeHtml(episode.name)}</p>
                <p class="player-hint">
                    <i data-lucide="wrench" class="icon"></i>
                    Adicione o link em <code>CONFIG.EPISODE_LINKS[${seasonNumber}][${episodeNumber}]</code>
                </p>
            </div>
        `;
        refreshIcons();
        showToast('Link deste episódio ainda não foi configurado', 'error');
        dom.playerContainer.scrollIntoView({ behavior: 'smooth', block: 'center' });
        return;
    }

    dom.playerPlaceholder.style.display = 'none';

    // Remove player anterior
    const existingIframe = dom.playerContainer.querySelector('iframe');
    if (existingIframe) existingIframe.remove();
    const existingVideo = dom.playerContainer.querySelector('video');
    if (existingVideo) existingVideo.remove();

    // Destroi instância HLS anterior
    if (state.hlsPlayer) {
        try { state.hlsPlayer.destroy(); } catch (e) {}
        state.hlsPlayer = null;
    }

    // Detecta se é HLS (.m3u8)
    const isHls = videoUrl.includes('.m3u8');

    if (isHls) {
        // 🎥 Cria elemento <video> com controles
        const video = document.createElement('video');
        video.controls = true;
        video.autoplay = true;
        video.playsInline = true;
        video.setAttribute('controlsList', 'nodownload noplaybackrate noremoteplayback');
        video.setAttribute('disablePictureInPicture', 'true');
        video.setAttribute('controlslist', 'nodownload');
        video.oncontextmenu = (e) => e.preventDefault();
        dom.playerContainer.appendChild(video);

        // Tenta usar HLS.js
        if (window.Hls && window.Hls.isSupported()) {
            const hls = new window.Hls({
                enableWorker: true,
                lowLatencyMode: false
            });
            hls.loadSource(videoUrl);
            hls.attachMedia(video);
            hls.on(window.Hls.Events.MANIFEST_PARSED, function() {
                video.play().catch(() => {});
            });
            hls.on(window.Hls.Events.ERROR, function(event, data) {
                if (data.fatal) {
                    console.error('HLS fatal error:', data);
                    showToast('Erro ao carregar vídeo', 'error');
                }
            });
            state.hlsPlayer = hls;
        }
        // Safari suporta HLS nativo
        else if (video.canPlayType('application/vnd.apple.mpegurl')) {
            video.src = videoUrl;
            video.addEventListener('loadedmetadata', function() {
                video.play().catch(() => {});
            });
        }
        // Fallback: tenta carregar direto
        else {
            video.src = videoUrl;
        }
    } else {
        // 🎬 Para URLs não-HLS, usa iframe
        const iframe = document.createElement('iframe');
        iframe.src = videoUrl;
        iframe.allowFullscreen = true;
        iframe.allow = 'autoplay; encrypted-media; fullscreen; picture-in-picture';
        iframe.setAttribute('frameborder', '0');
        iframe.setAttribute('sandbox', 'allow-scripts allow-same-origin allow-presentation allow-forms allow-popups');
        dom.playerContainer.appendChild(iframe);
    }

    dom.playerContainer.classList.add('playing');

    showToast(`Reproduzindo T${seasonNumber}E${episodeNumber}`, 'success');
    dom.playerContainer.scrollIntoView({ behavior: 'smooth', block: 'center' });
}

async function init() {
    refreshIcons();

    if (!CONFIG.API_KEY || CONFIG.API_KEY === 'COLE_SUA_API_KEY_AQUI') {
        dom.tvTitle.textContent = 'API Key necessária';
        dom.tvOverview.textContent = 'Configure sua API Key no código (CONFIG.API_KEY).';
        showToast('API Key não configurada', 'error');
        return;
    }

    showLoading('Carregando Overflow: Transbordando...');

    try {
        const data = await fetchTVShow(state.tvId);
        renderTVShow(data);
        await loadSeason(1);
    } catch (error) {
        console.error('Erro ao carregar série:', error);
        dom.tvTitle.textContent = 'Erro ao carregar';
        dom.tvOverview.textContent = error.message;
        showToast(error.message, 'error');
    } finally {
        hideLoading();
    }
}
